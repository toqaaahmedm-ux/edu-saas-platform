"use client";

import { useParams } from "next/navigation";
import { Loader2, Award } from "lucide-react";
import { useCourseGrades, useRecomputeGrade } from "@/services/grades.service";
import { toast } from "sonner";

export default function TeacherGradesPage() {
  const { courseId } = useParams<{ courseId: string }>();
  const { data: grades = [], isLoading, isError } = useCourseGrades(courseId);
  const { mutate: recompute, isPending } = useRecomputeGrade(courseId);

  if (isLoading) {
    return (
      <div className="flex justify-center p-20">
        <Loader2 className="animate-spin text-blue-600" size={40} />
      </div>
    );
  }

  if (isError) {
    return (
      <div className="text-center py-20 border-2 border-dashed border-red-200 rounded-3xl">
        <p className="text-red-500 font-medium">Failed to load grades.</p>
      </div>
    );
  }

  return (
    <div className="p-8">
      <h1 className="text-3xl font-black mb-8">Grades</h1>

      {grades.length > 0 ? (
        <div className="bg-white rounded-3xl border border-slate-100 shadow-sm overflow-hidden">
          <table className="w-full text-sm">
            <thead className="bg-slate-50 text-slate-400 text-xs uppercase tracking-widest">
              <tr>
                <th className="text-left px-6 py-3">Student</th>
                <th className="text-left px-6 py-3">Score</th>
                <th className="text-left px-6 py-3">Letter</th>
                <th className="text-left px-6 py-3">GPA</th>
                <th className="px-6 py-3"></th>
              </tr>
            </thead>
            <tbody>
              {grades.map((grade: any) => (
                <tr key={grade.id} className="border-t border-slate-100">
                  <td className="px-6 py-4">
                    <p className="font-bold">{grade.student?.name}</p>
                    <p className="text-slate-400 text-xs">{grade.student?.email}</p>
                  </td>
                  <td className="px-6 py-4 font-black">{grade.score}%</td>
                  <td className="px-6 py-4">
                    <span className="bg-blue-50 text-blue-700 font-black px-2 py-1 rounded-lg text-xs">
                      {grade.letterGrade || "—"}
                    </span>
                  </td>
                  <td className="px-6 py-4">{grade.gpa ?? "—"}</td>
                  <td className="px-6 py-4 text-right">
                    <button
                      disabled={isPending}
                      onClick={() =>
                        recompute(grade.studentId, {
                          onSuccess: () => toast.success("Grade recalculated"),
                          onError: () => toast.error("Failed to recalculate"),
                        })
                      }
                      className="text-blue-600 text-xs font-bold hover:underline disabled:opacity-50"
                    >
                      Recalculate
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      ) : (
        <div className="text-center py-20 border-2 border-dashed border-slate-200 rounded-3xl">
          <Award size={32} className="mx-auto text-slate-300 mb-3" />
          <p className="text-slate-500 font-medium">
            No grades computed yet. Grades appear automatically once you grade an assignment.
          </p>
        </div>
      )}
    </div>
  );
}
